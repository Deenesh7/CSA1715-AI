teaches(dr_kumar, cs101).
teaches(dr_kumar, cs102).
teaches(dr_meena, cs103).
teaches(dr_ravi,  ma101).

% studies(Student, SubjectCode).
studies(rahul,  cs101).
studies(rahul,  cs103).
studies(priya,  cs102).
studies(priya,  ma101).
studies(arjun,  cs101).
studies(arjun,  cs102).
studies(sneha,  cs103).
studies(sneha,  ma101).

% Teacher of a student (student studies a subject the teacher teaches)
taught_by(Student, Teacher) :-
    studies(Student, Code),
    teaches(Teacher, Code).

% Students of a teacher
teacher_of_student(Teacher, Student) :-
    taught_by(Student, Teacher).

% Subject name studied by a student
student_subject_name(Student, SubjectName) :-
    studies(Student, Code),
    subject(Code, SubjectName).

% Two students are classmates if they share a subject
classmates(S1, S2) :-
    studies(S1, Code),
    studies(S2, Code),
    S1 \= S2.
