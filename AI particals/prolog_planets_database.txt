% Planet type
type(mercury, terrestrial).
type(venus,   terrestrial).
type(earth,   terrestrial).
type(mars,    terrestrial).
type(jupiter, gas_giant).
type(saturn,  gas_giant).
type(uranus,  ice_giant).
type(neptune, ice_giant).

% Planets that have moons
has_moons(P) :-
    planet(P, _, _, M),
    M > 0.

% Planet bigger than another one
bigger(P1, P2) :-
    planet(P1, _, D1, _),
    planet(P2, _, D2, _),
    D1 > D2.

% Planet is closer to the Sun than another
closer_to_sun(P1, P2) :-
    planet(P1, O1, _, _),
    planet(P2, O2, _, _),
    O1 < O2.

% Planets of a given type
planets_of_type(Type, P) :-
    type(P, Type).
