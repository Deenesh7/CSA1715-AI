% Prolog: Database with Name and Date of Birth
% dob(Name, date(Day, Month, Year)).

dob(rahul,  date(15, 3, 2002)).
dob(priya,  date(22, 7, 2001)).
dob(arjun,  date(9, 11, 2003)).
dob(sneha,  date(30, 1, 2002)).
dob(karthik, date(5, 6, 2000)).

% Born in a given year
born_in_year(Name, Year) :-
    dob(Name, date(_, _, Year)).

% Born in a given month
born_in_month(Name, Month) :-
    dob(Name, date(_, Month, _)).

% Person A is older than person B
older(A, B) :-
    dob(A, date(D1, M1, Y1)),
    dob(B, date(D2, M2, Y2)),
    A \= B,
    ( Y1 < Y2
    ; Y1 =:= Y2, M1 < M2
    ; Y1 =:= Y2, M1 =:= M2, D1 < D2
    ).
